## Q6. AJAX: fetch JSON and display it

```html
<!DOCTYPE html>
<html>
<head>
<title>Q6</title>
<style>
  body { text-align:center; }
  table { border-collapse:collapse; margin:20px auto; }
  td, th { border:1px solid gray; padding:6px; }
  th { background:#0d6efd; color:white; }
</style>
</head>
<body>
  <button onclick="load()">Load Users</button>
  <table id="data"></table>

<script>
function load() {
  var x = new XMLHttpRequest();
  x.open("GET", "https://jsonplaceholder.typicode.com/users");
  x.onload = function () {
    var rows = "<tr><th>ID</th><th>Name</th><th>Email</th></tr>";
    JSON.parse(x.responseText).forEach(function (u) {
      rows += "<tr><td>" + u.id + "</td><td>" + u.name + "</td><td>" + u.email + "</td></tr>";
    });
    document.getElementById("data").innerHTML = rows;
  };
  x.send();
}
</script>
</body>
</html>
```

## Q7. JavaScript form validation

```html
<!DOCTYPE html>
<html>
<head>
<title>Q7</title>
<style>
  form { width:250px; margin:40px auto; }
  input { display:block; width:100%; margin:6px 0; padding:6px; box-sizing:border-box; }
  #msg { color:red; }
</style>
</head>
<body>
<form onsubmit="return check()">
  <input id="name" placeholder="Name">
  <input id="email" placeholder="Email">
  <input id="pass" type="password" placeholder="Password">
  <button>Submit</button>
  <p id="msg"></p>
</form>

<script>
function check() {
  var n = document.getElementById("name").value.trim();
  var e = document.getElementById("email").value.trim();
  var p = document.getElementById("pass").value;
  if (n === "" || e === "" || p === "") {
    document.getElementById("msg").innerHTML = "All fields are required!";
    return false;      // stops the form from submitting
  }
  alert("Form submitted successfully");
  return true;
}
</script>
</body>
</html>
```

## Q11. Node.js + Express server

Terminal:
```bash
npm init -y
npm install express
node server.js
```
Open `http://localhost:3000`.

**server.js**
```js
const express = require("express");
const app = express();
app.use(express.json());

app.get("/", (req, res) => {
  res.send("<style>body{font-family:Arial;text-align:center;background:#eef2f7}h1{color:green}</style>" +
           "<h1>Hello from Express!</h1><a href='/api/users'>View Users</a>");
});

app.get("/api/users", (req, res) => {
  res.json([{ id: 1, name: "Asha" }, { id: 2, name: "Ravi" }]);
});

app.post("/api/users", (req, res) => {
  res.status(201).json(req.body);
});

app.listen(3000, () => console.log("Server at http://localhost:3000"));
```

## Q12. Node.js + MongoDB CRUD

Start MongoDB first (check with `mongosh`). Terminal:
```bash
npm init -y
npm install mongodb
node crud.js
```

**crud.js**
```js
const { MongoClient } = require("mongodb");
const client = new MongoClient("mongodb://127.0.0.1:27017");

async function main() {
  await client.connect();
  const col = client.db("college").collection("students");

  await col.deleteMany({});                                                       // clean start
  await col.insertMany([{ name: "Asha", marks: 85 }, { name: "Ravi", marks: 72 }]); // CREATE
  console.log(await col.find().toArray());                                          // READ
  await col.updateOne({ name: "Asha" }, { $set: { marks: 95 } });                   // UPDATE
  await col.deleteOne({ name: "Ravi" });                                            // DELETE
  console.log(await col.find().toArray());

  await client.close();
}
main();
```

This one prints to the console, so there is no CSS. The Q11 server already shows how a styled page is served.
