```html

      background: lightblue;
    }
```

## Q4. JavaScript: take user input and display the result dynamically

```html
<!DOCTYPE html>
<html>

  <title>Student Result</title>
  <style>
    body { font-family: Arial; background: #f0f4f8; text-align: center; padding-top: 50px; }
    .box { background: white; display: inline-block; padding: 30px; border-radius: 10px; box-shadow: 0 0 10px gray; }

    #output { margin-top: 15px; font-size: 18px; }
  </style>
</head>
<body>
  <div class="box">
    <h2>Student Result</h2>
    <input type="text" id="name" placeholder="Enter name"><br>
    <input type="number" id="marks" placeholder="Enter marks"><br>
    <button onclick="showResult()">Show Result</button>
    <p id="output"></p>
  </div>

  <script>
    function showResult() {
      var name = document.getElementById("name").value;
      var marks = document.getElementById("marks").value;

      if (name === "" || marks === "") {
        document.getElementById("output").innerHTML = "Please enter both values";
        return;
      }

      var result = Number(marks) >= 40 ? "Pass" : "Fail";
      document.getElementById("output").innerHTML =
        name + " scored " + marks + " marks. Result: <b>" + result + "</b>";
    }
  </script>
</body>
</html>
```

**Remember:** `document.getElementById(...).value` reads input, and `.innerHTML` writes output.

## Q6. Fetch data using AJAX and display JSON on the web page

Uses classic `XMLHttpRequest` (the original AJAX) and a free test API.

```html
<!DOCTYPE html>
<html>
<head>
  <title>AJAX JSON Demo</title>
  <style>
    body { font-family: Arial; text-align: center; padding: 30px; }
    button { padding: 10px 25px; background: #198754; color: white; border: none; border-radius: 5px; cursor: pointer; }
    table { margin: 20px auto; border-collapse: collapse; width: 80%; }
    th, td { border: 1px solid #999; padding: 8px; }
    th { background: #0d6efd; color: white; }
    tr:nth-child(even) { background: #f2f2f2; }
  </style>
</head>
<body>
  <h2>Users from JSON (AJAX)</h2>
  <button id="load">Load Users</button>

  <table>
    <thead>
      <tr><th>ID</th><th>Name</th><th>Email</th><th>City</th></tr>
    </thead>
    <tbody id="data"></tbody>
  </table>

  <script>
    document.getElementById("load").addEventListener("click", function () {
      var xhr = new XMLHttpRequest();
      xhr.open("GET", "https://jsonplaceholder.typicode.com/users", true);

      xhr.onreadystatechange = function () {
        if (xhr.readyState === 4 && xhr.status === 200) {
          var users = JSON.parse(xhr.responseText);
          var rows = "";
          users.forEach(function (u) {
            rows += "<tr><td>" + u.id + "</td><td>" + u.name + "</td><td>" +
                    u.email + "</td><td>" + u.address.city + "</td></tr>";
          });
          document.getElementById("data").innerHTML = rows;
        }
      };

      xhr.send();
    });
  </script>
</body>
</html>
```

**Shorter alternative using `fetch`** (replace the script body if your teacher uses it):

```js
fetch("https://jsonplaceholder.typicode.com/users")
  .then(res => res.json())
  .then(users => {
    let rows = "";
    users.forEach(u => {
      rows += `<tr><td>${u.id}</td><td>${u.name}</td><td>${u.email}</td><td>${u.address.city}</td></tr>`;
    });
    document.getElementById("data").innerHTML = rows;
  });
```

**Remember:** `readyState === 4` means the request is done, and `status === 200` means OK. `JSON.parse` turns text into an object. If you use a local `data.json` file instead, open the page with the VS Code **Live Server** extension, because AJAX does not work from a plain `file://` path.

## Q7. JavaScript form validation (required fields not empty)

```html
<!DOCTYPE html>
<html>
<head>
  <title>Form Validation</title>
  <style>
    body { font-family: Arial; background: #f0f4f8; }
    form { width: 320px; margin: 50px auto; background: white; padding: 25px; border-radius: 10px; box-shadow: 0 0 10px gray; }
    input { width: 100%; padding: 8px; margin: 5px 0; box-sizing: border-box; }
    .error { color: red; font-size: 13px; }
    button { width: 100%; padding: 10px; background: #0d6efd; color: white; border: none; border-radius: 5px; cursor: pointer; margin-top: 10px; }
  </style>
</head>
<body>
  <form onsubmit="return validateForm()">
    <h2>Register</h2>

    <label>Name</label>
    <input type="text" id="name">
    <span class="error" id="nameErr"></span><br>

    <label>Email</label>
    <input type="text" id="email">
    <span class="error" id="emailErr"></span><br>

    <label>Password</label>
    <input type="password" id="pass">
    <span class="error" id="passErr"></span><br>

    <button type="submit">Submit</button>
  </form>

  <script>
    function validateForm() {
      var valid = true;
      var name  = document.getElementById("name").value.trim();
      var email = document.getElementById("email").value.trim();
      var pass  = document.getElementById("pass").value;

      // clear old messages
      document.getElementById("nameErr").innerText = "";
      document.getElementById("emailErr").innerText = "";
      document.getElementById("passErr").innerText = "";

      if (name === "") {
        document.getElementById("nameErr").innerText = "Name is required";
        valid = false;
      }
      if (email === "") {
        document.getElementById("emailErr").innerText = "Email is required";
        valid = false;
      } else if (email.indexOf("@") === -1) {
        document.getElementById("emailErr").innerText = "Enter a valid email";
        valid = false;
      }
      if (pass === "") {
        document.getElementById("passErr").innerText = "Password is required";
        valid = false;
      } else if (pass.length < 6) {
        document.getElementById("passErr").innerText = "Minimum 6 characters";
        valid = false;
      }

      if (valid) { alert("Form submitted successfully!"); }
      return valid;   // false stops the form from submitting
    }
  </script>
</body>
</html>
```

**Remember:** `onsubmit="return validateForm()"`, and returning `false` blocks submission.

## Q8. JavaScript arrays and objects: store and display values

```html
<!DOCTYPE html>
<html>
<head>
  <title>Arrays and Objects</title>
  <style>
    body { font-family: Arial; text-align: center; padding: 30px; }
    table { margin: 20px auto; border-collapse: collapse; width: 60%; }
    th, td { border: 1px solid #555; padding: 8px; }
    th { background: #6f42c1; color: white; }
    .info { background: #e9ecef; display: inline-block; padding: 10px 20px; border-radius: 8px; }
  </style>
</head>
<body>
  <h2>Student Details</h2>

  <table>
    <thead><tr><th>ID</th><th>Name</th><th>Dept</th><th>Marks</th></tr></thead>
    <tbody id="tbody"></tbody>
  </table>

  <p class="info" id="summary"></p>
  <p>Subjects (array): <span id="subjects"></span></p>

  <script>
    // Array of objects
    var students = [
      { id: 1, name: "Asha",  dept: "CSE", marks: 85 },
      { id: 2, name: "Ravi",  dept: "ECE", marks: 72 },
      { id: 3, name: "Meena", dept: "CSE", marks: 91 }
    ];

    // add a new object to the array
    students.push({ id: 4, name: "Kiran", dept: "IT", marks: 66 });

    // display with a loop
    var rows = "", total = 0;
    for (var i = 0; i < students.length; i++) {
      rows += "<tr><td>" + students[i].id + "</td><td>" + students[i].name +
              "</td><td>" + students[i].dept + "</td><td>" + students[i].marks + "</td></tr>";
      total += students[i].marks;
    }
    document.getElementById("tbody").innerHTML = rows;
    document.getElementById("summary").innerHTML =
      "Total students: " + students.length + " | Average marks: " + (total / students.length).toFixed(2);

    // simple array of strings
    var subjects = ["HTML", "CSS", "JavaScript", "React"];
    document.getElementById("subjects").innerText = subjects.join(", ");
  </script>
</body>
</html>
```

**Remember:** an array uses `[ ]` and an object uses `{ key: value }`. Use `.push()` to add, `.length` to count, and `obj.key` to read a value.

## Q10. React component using JSX and props to display user details

Same CDN setup as Q9. Only the `<style>` and the script change.

```html
<!DOCTYPE html>
<html>

  <meta charset="UTF-8">
  <title>User Details</title>
  <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>

  </style>
</head>
<body>
  <div id="root"></div>

  <script type="text/babel">
    // Child component: receives data through props
    function UserCard(props) {
      return (
        <div className="user-card">
          <h3>{props.name}</h3>
          <p>Email: {props.email}</p>
          <p>City: {props.city}</p>
          <p>Age: {props.age}</p>
        </div>
      );
    }

    // Parent component: passes props to the child
    function App() {
      const users = [
        { id: 1, name: "Asha",  email: "asha@mail.com",  city: "Bengaluru", age: 21 },
        { id: 2, name: "Ravi",  email: "ravi@mail.com",  city: "Mysuru",    age: 22 },
        { id: 3, name: "Meena", email: "meena@mail.com", city: "Chennai",   age: 20 }
      ];

      return (
        <div>
          <h1>User Details</h1>
          <div className="list">
            {users.map((u) => (
              <UserCard key={u.id} name={u.name} email={u.email} city={u.city} age={u.age} />
            ))}
          </div>
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById("root")).render(<App />);
  </script>
</body>
</html>
```

**Remember:** props are read-only data passed from parent to child, like `<UserCard name="Asha" />`, and read as `props.name`. `.map()` loops through an array, and each item needs a unique `key`.

## Q11. Server using Node.js and Express.js (handle HTTP requests, return responses)

**Folder structure:**
```
q11/
├── server.js
└── public/
    ├── index.html
    └── style.css
```

**Terminal commands (run inside the `q11` folder):**
```bash
npm init -y
npm install express
node server.js
```
Then open `http://localhost:3000` in the browser.

**server.js**
```js
const express = require("express");
const app = express();

app.use(express.json());          // lets the server read JSON sent in POST requests
app.use(express.static("public")); // serves index.html and style.css at "/"

let users = [
  { id: 1, name: "Asha", city: "Bengaluru" },
  { id: 2, name: "Ravi", city: "Mysuru" }
];

// GET: simple text response
app.get("/hello", (req, res) => {
  res.send("Hello from Express server!");
});

// GET: return JSON list
app.get("/api/users", (req, res) => {
  res.json(users);
});

// GET: one user by id
app.get("/api/users/:id", (req, res) => {
  const user = users.find(u => u.id === Number(req.params.id));
  if (!user) return res.status(404).json({ message: "User not found" });
  res.json(user);
});

// POST: add a user
app.post("/api/users", (req, res) => {
  const newUser = { id: users.length + 1, name: req.body.name, city: req.body.city };
  users.push(newUser);
  res.status(201).json(newUser);
});

app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});
```

## Q12. Node.js app connected to MongoDB: CRUD operations on a collection

**Terminal commands:**
```bash

```
MongoDB must be running first. Check with `mongosh`.

**crud.js**
```js
const { MongoClient } = require("mongodb");

const url = "mongodb://127.0.0.1:27017";
const client = new MongoClient(url);

async function main() {
  await client.connect();
  console.log("Connected to MongoDB");

  const db = client.db("college");
  const students = db.collection("students");

  await students.deleteMany({});   // start clean so re-running does not duplicate data

  // ---------- CREATE ----------
  await students.insertOne({ name: "Asha", dept: "CSE", marks: 85 });
  await students.insertMany([
    { name: "Ravi",  dept: "ECE", marks: 72 },
    { name: "Meena", dept: "CSE", marks: 91 }
  ]);
  console.log("\n--- After INSERT ---");
  console.log(await students.find().toArray());

  // ---------- READ ----------
  console.log("\n--- READ: CSE students ---");
  console.log(await students.find({ dept: "CSE" }).toArray());

  // ---------- UPDATE ----------
  await students.updateOne({ name: "Asha" }, { $set: { marks: 95 } });
  console.log("\n--- After UPDATE (Asha marks = 95) ---");
  console.log(await students.find().toArray());

  // ---------- DELETE ----------
  await students.deleteOne({ name: "Ravi" });
  console.log("\n--- After DELETE (Ravi removed) ---");
  console.log(await students.find().toArray());
}

main()
  .catch(console.error)
  .finally(() => client.close());
```

**Remember:** Create is `insertOne`/`insertMany`, Read is `find`, Update is `updateOne` with `$set`, and Delete is `deleteOne`. A database holds collections, and a collection holds documents (like rows). This version prints to the console, so there is no CSS here. If the examiner wants a web page, use the Q11 `public/` folder with the version below.

### Alternative: Express + Mongoose REST API

```js
const express = require("express");
const mongoose = require("mongoose");

const app = express();
app.use(express.json());
app.use(express.static("public"));

mongoose.connect("mongodb://127.0.0.1:27017/college")
  .then(() => console.log("MongoDB connected"))
  .catch(err => console.log(err));

const Student = mongoose.model("Student", new mongoose.Schema({
  name: String,
  dept: String,
  marks: Number
}));

app.post("/students", async (req, res) => res.json(await Student.create(req.body)));          // Create
app.get("/students", async (req, res) => res.json(await Student.find()));                      // Read
app.put("/students/:id", async (req, res) =>
  res.json(await Student.findByIdAndUpdate(req.params.id, req.body, { new: true })));          // Update
app.delete("/students/:id", async (req, res) => {
  await Student.findByIdAndDelete(req.params.id);
  res.json({ message: "Deleted" });                                                              // Delete
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```
