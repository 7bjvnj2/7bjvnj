 3)KNN 

import matplotlib.pyplot as plt
from sklearn.neighbors import KNeighborsClassifier

# Training data: Study Hours, Sleep Hours
X = [
    [8, 8],
    [7, 7],
    [9, 8],
    [1, 4],
    [2, 5],
    [3, 4],
    [8, 6],
    [2, 3]
]

# Output labels
y = [
    "Pass", "Pass", "Pass", "Fail",
    "Fail", "Fail", "Pass", "Fail"
]

# Train the KNN model
model = KNeighborsClassifier(n_neighbors=3)
model.fit(X, y)

# Get user input
study = float(input("Enter study hours: "))
sleep = float(input("Enter sleep hours: "))

# Predict student result
prediction = model.predict([[study, sleep]])[0]

print("Student Result:", prediction.upper())

# Plot training data
for i in range(len(X)):
    plt.scatter(
        X[i][0],
        X[i][1],
        color="green" if y[i] == "Pass" else "red",
        label=y[i] if y[i] not in [
            y[j] for j in range(i)
        ] else ""
    )

# Plot user input
plt.scatter(
    study,
    sleep,
    color="blue",
    marker="*",
    s=250,
    label="New Student"
)

# Graph labels and title
plt.xlabel("Study Hours")
plt.ylabel("Sleep Hours")
plt.title("KNN Classification - Pass or Fail")

plt.legend()
plt.grid(True)

# Display graph
plt.show()

4) LINEAR REGRESSION

import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

# Training data (Seen Data)
X = [[500], [700], [900], [1100], [1300], [1500]]
y = [20, 28, 36, 44, 52, 60]  # Price in Lakhs

# Create and train the model
model = LinearRegression()
model.fit(X, y)

# Get user input (Unseen Data)
area = float(input("Enter house area in sq. ft.: "))

# Predict house price
prediction = model.predict([[area]])

print("Predicted House Price:", round(prediction[0], 2), "Lakhs")

# Plot training data
plt.scatter(
    [row[0] for row in X],
    y,
    color="green",
    label="Training Data"
)

# Plot new house prediction
plt.scatter(
    area,
    prediction[0],
    color="red",
    marker="*",
    s=200,
    label="New House"
)

# Plot regression line
plt.plot(
    [row[0] for row in X],
    model.predict(X),
    color="blue",
    label="Regression Line"
)

# Graph labels and title
plt.xlabel("House Area (sq. ft.)")
plt.ylabel("House Price (Lakhs)")
plt.title("Linear Regression - House Price Prediction")

plt.legend()
plt.grid(True)

# Display graph
plt.show()

9) MULTIPLE REGRESSION

import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

# Training data (Seen Data)
# Features: Area, Bedrooms, Parking (1 = Yes, 0 = No)

X = [
    [500, 1, 0],
    [700, 2, 1],
    [900, 2, 0],
    [1100, 3, 1],
    [1300, 3, 1],
    [1500, 4, 1]
]

# House prices in Lakhs
y = [20, 30, 32, 48, 55, 65]

# Create and train the model
model = LinearRegression()
model.fit(X, y)

# Get user input (Unseen Data)
area = float(input("Enter house area in sq. ft.: "))
bedrooms = int(input("Enter number of bedrooms: "))
parking = int(input("Parking available? (1 = Yes, 0 = No): "))

# Validate parking input
if parking not in [0, 1]:
    print("Invalid parking input! Enter 1 or 0.")
else:
    # Predict house price
    prediction = model.predict([[area, bedrooms, parking]])

    print(
        "Predicted House Price:",
        round(prediction[0], 2),
        "Lakhs"
    )

    # Plot training data
    plt.scatter(
        [row[0] for row in X],
        y,
        color="green",
        label="Training Data"
    )

    # Plot new house prediction
    plt.scatter(
        area,
        prediction[0],
        color="red",
        marker="*",
        s=200,
        label="New House"
    )

    # Graph labels and title
    plt.xlabel("House Area (sq. ft.)")
    plt.ylabel("House Price (Lakhs)")
    plt.title("Multiple Linear Regression - House Price Prediction")

    plt.legend()
    plt.grid(True)

    # Display graph
    plt.show()
