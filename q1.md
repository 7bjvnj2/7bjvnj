## 1. Best First Search

```python
import heapq
graph = {'S': {'A': 1, 'B': 4}, 'A': {'C': 2}, 'B': {'G': 12}, 'C': {'G': 3}, 'G': {}}
h = {'S': 6, 'A': 5, 'B': 2, 'C': 3, 'G': 0}
pq = [(h['S'], 0, 'S', ['S'])]
seen = set()
while pq:
    f, g, n, path = heapq.heappop(pq)
    if n == 'G':
        print(path, g)
        break
    if n in seen:
        continue
    seen.add(n)
    for m, c in graph[n].items():
        heapq.heappush(pq, (h[m], g + c, m, path + [m]))
```

## 2. MinMax Algorithm

```python
tree = [[[3, 5], [6, 9]], [[1, 2], [0, -1]]]

def minimax(n, mx):
    if isinstance(n, int):
        return n
    v = [minimax(c, not mx) for c in n]
    return max(v) if mx else min(v)

print(minimax(tree, True))
```

## 5. Water Jug

```python
from collections import deque
a, b, t = 4, 3, 2
q = deque([((0, 0), [(0, 0)])])
seen = {(0, 0)}
while q:
    (x, y), p = q.popleft()
    if t in (x, y):
        print(p)
        break
    for s in [(a, y), (x, b), (0, y), (x, 0),
              (x - min(x, b - y), y + min(x, b - y)),
              (x + min(y, a - x), y - min(y, a - x))]:
        if s not in seen:
            seen.add(s)
            q.append((s, p + [s]))
```

## 8. AO* Algorithm

```python
graph = {'A': [['B', 'C'], ['D']], 'B': [['E'], ['F']], 'C': [['G', 'H']], 'D': [['I']]}
h = {'E': 4, 'F': 2, 'G': 1, 'H': 2, 'I': 9}
sol = {}

def ao(n):
    if n not in graph:
        return h[n]
    cost, sol[n] = min((sum(1 + ao(c) for c in o), o) for o in graph[n])
    return cost

def show(n):
    if n in sol:
        print(n, '->', sol[n])
        for c in sol[n]:
            show(c)

print(ao('A'))
show('A')
```

## 9. Alpha Beta Pruning

```python
from math import inf
tree = [[[3, 5], [6, 9]], [[1, 2], [0, -1]]]

def ab(n, mx, a=-inf, b=inf):
    if isinstance(n, int):
        return n
    for c in n:
        v = ab(c, not mx, a, b)
        if mx:
            a = max(a, v)
        else:
            b = min(b, v)
        if a >= b:
            break
    return a if mx else b

print(ab(tree, True))
```

## 10. N-Queens

```python
N = 4

def solve(p=()):
    r = len(p)
    if r == N:
        return p
    for c in range(N):
        if all(c != x and abs(c - x) != r - i for i, x in enumerate(p)):
            s = solve(p + (c,))
            if s:
                return s

for c in solve():
    print(' '.join('Q' if i == c else '.' for i in range(N)))
```

## 11. Hill Climbing

f = lambda x: -(x - 3) ** 2
x = 0
while f(x + 1) > f(x) or f(x - 1) > f(x):
    x = max(x - 1, x + 1, key=f)
